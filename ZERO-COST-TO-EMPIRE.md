### Single-File Brain Definition

```python
# brains/content_trend_analyzer.py
import anthropic
import json
from datetime import datetime

class TrendAnalyzerBrain:
    def __init__(self):
        self.client = anthropic.Anthropic()  # Uses ANTHROPIC_API_KEY env var
        self.brain_id = "content_trend_analyzer"
    
    def run(self):
        """Analyze trending topics on YouTube/TikTok"""
        prompt = """
        Analyze the top 10 trending topics on YouTube right now.
        For each, suggest a 60-second video concept that would perform well.
        Format as JSON: [{"topic": "...", "concept": "..."}]
        """
        
        message = self.client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=1024,
            messages=[{"role": "user", "content": prompt}]
        )
        
        return {
            "brain_id": self.brain_id,
            "output": json.loads(message.content[0].text),
            "timestamp": datetime.now().isoformat(),
            "cost_estimate": "~$0.10"  # Rough estimate
        }

# That's it. One brain = ~30 lines of code.
# Deploy 100 brains = 3,000 lines of Python.
```

```bash
import anthropic
import supabase
import json
from datetime import datetime
import os

class BaseBrain:
    def __init__(self, brain_id: str, brain_name: str):
        self.brain_id = brain_id
        self.brain_name = brain_name
        self.client = anthropic.Anthropic(api_key=os.getenv('ANTHROPIC_API_KEY'))
        self.db = supabase.create_client(
            os.getenv('SUPABASE_URL'),
            os.getenv('SUPABASE_KEY')
        )
    
    def think(self, prompt: str) -> str:
        """Think using Claude"""
        message = self.client.messages.create(
            model="claude-3-5-sonnet-20241022",
            max_tokens=2048,
            messages=[{"role": "user", "content": prompt}]
        )
        return message.content[0].text
    
    def store_output(self, output: dict, cost_estimate: float = 0.0):
        """Store brain output in database"""
        self.db.table('brain_outputs').insert({
            'brain_id': self.brain_id,
            'output': output,
            'created_at': datetime.now().isoformat(),
            'cost_estimate': cost_estimate
        })
    
    def get_context(self, query: str) -> list:
        """Retrieve previous outputs for learning"""
        response = self.db.table('brain_outputs')\
            .filter('brain_id', 'eq', self.brain_id)\
            .select('output')\
            .limit(10)\
            .execute()
        return response['data']
    
    def log(self, message: str):
        """Log to console and database"""
        print(f"[{self.brain_name}] {message}")

if __name__ == "__main__":
    brain = BaseBrain("test", "Test Brain")
    result = brain.think("What is 2+2?")
    print(result)
EOF
```
